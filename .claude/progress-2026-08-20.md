```javascript
// ================= OCR 文字识别 =================
bindDropzone(ocrDrop, ocrInput, false, function (files) {
    if (!files.length) return;
    var f = files[0];
    if (f.type !== 'application/pdf' && !/\.pdf$/i.test(f.name)) { 
        showToast('请选择 PDF 文件'); 
        return; 
    }
    ocrFile = f;
    ocrOpts.style.display = 'flex';
    ocrResult.style.display = 'none';
    ocrBtn.disabled = false;
    
    // 读取页数
    fileToArrayBuffer(f).then(function (buf) {
        return PDFLib.PDFDocument.load(buf, { ignoreEncryption: true });
    }).then(function (doc) {
        ocrTotal.textContent = doc.getPageCount();
    }).catch(function (err) {
        showToast('无法读取 PDF');
        console.error(err);
    });
});

ocrBtn.addEventListener('click', function () {
    if (!ocrFile) { showToast('请先选择 PDF'); return; }
    
    requireAdminPassword(function() {
        doOCR();
    });
});

function doOCR() {
    setBtnLoading(ocrBtn, '识别中...');
    
    var formData = new FormData();
    formData.append('file', ocrFile);
    formData.append('pages', ocrRange.value.trim());
    formData.append('lang', ocrLang.value);
    
    fetch('/api/pdf/ocr', {
        method: 'POST',
        headers: { 'X-Admin-Password': adminPassword },
        body: formData
    })
    .then(function (resp) {
        if (resp.status === 401) {
            adminPassword = null;
            throw new Error('密码错误，请重试');
        }
        if (!resp.ok) throw new Error('识别失败');
        return resp.json();
    })
    .then(function (data) {
        if (!data.ok) throw new Error(data.error || '识别失败');
        
        var text = data.results.map(function(r) {
            return '=== 第 ' + r.pageNo + ' 页 ===\n' + r.text;
        }).join('\n\n');
        
        ocrText.value = text;
        ocrResult.style.display = 'flex';
        showToast('识别完成');
        setBtnLoading(ocrBtn, '开始识别');
    })
    .catch(function (err) {
        setBtnLoading(ocrBtn, '开始识别');
        showToast(err.message || '识别失败');
        console.error(err);
    });
}

ocrCopy.addEventListener('click', function() {
    ocrText.select();
    document.execCommand('copy');
    showToast('已复制到剪贴板');
});

// ================= PDF 加密 =================
bindDropzone(encryptDrop, encryptInput, false, function (files) {
    if (!files.length) return;
    var f = files[0];
    if (f.type !== 'application/pdf' && !/\.pdf$/i.test(f.name)) { 
        showToast('请选择 PDF 文件'); 
        return; 
    }
    encryptFile = f;
    encryptOpts.style.display = 'flex';
    encryptBtn.disabled = false;
});

encryptBtn.addEventListener('click', function () {
    if (!encryptFile) { showToast('请先选择 PDF'); return; }
    
    var password = encryptPassword.value.trim();
    if (!password) { showToast('请输入密码'); return; }
    
    requireAdminPassword(function() {
        doEncrypt(password);
    });
});

function doEncrypt(password) {
    setBtnLoading(encryptBtn, '加密中...');
    
    var formData = new FormData();
    formData.append('file', encryptFile);
    formData.append('password', password);
    
    fetch('/api/pdf/encrypt', {
        method: 'POST',
        headers: { 'X-Admin-Password': adminPassword },
        body: formData
    })
    .then(function (resp) {
        if (resp.status === 401) {
            adminPassword = null;
            throw new Error('管理员密码错误');
        }
        if (!resp.ok) throw new Error('加密失败');
        return resp.json();
    })
    .then(function (data) {
        if (!data.ok) throw new Error(data.error || '加密失败');
        
        var bytes = Uint8Array.from(atob(data.data), function(c) { 
            return c.charCodeAt(0); 
        });
        var blob = new Blob([bytes], { type: 'application/pdf' });
        downloadBlob(blob, 'encrypted.pdf');
        showToast('加密完成');
        setBtnLoading(encryptBtn, '加密并下载');
        encryptPassword.value = '';
    })
    .catch(function (err) {
        setBtnLoading(encryptBtn, '加密并下载');
        showToast(err.message || '加密失败');
        console.error(err);
    });
}
```

## 🐛 注意事项

1. **密码验证**：adminPassword 变量在用户首次使用后端功能时通过 prompt 获取
2. **Base64 解码**：后端返回的 PDF 是 base64，前端需要转换为 Blob
3. **错误处理**：401 状态码需清除 adminPassword，让用户重新输入
4. **文件清理**：后端会自动清理临时文件（每 10 分钟 + 超过 1 小时的文件）
5. **CORS**：后端在同域，无需配置 CORS
6. **OCR 限制**：一次最多识别 10 页（防止超时）
